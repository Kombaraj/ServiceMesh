# Install Istio CLI #

# Go to https://github.com/istio/istio/releases
# Pick a release
# Download and unpack the release, and move it to a directory set in the `PATH`
istioctl version

###############################
# Create Kubernetes Cluster #
Refer: setup-eks.txt
###############################

####################
# Install Istio #
####################

istioctl profile list

istioctl profile dump demo

istioctl install \
    --set profile=demo -y

kubectl get crds | grep 'istio.io'

kubectl --namespace istio-system \
    get pods,svc,deploy

# Confirm that `EXTERNAL-IP` of `istio-ingressgateway` is not `pending`, unless using minikube

############################
# Manual Sidecar Injection #
############################

cd code

cat common/alpine.yml

istioctl kube-inject \
    --filename common/alpine.yml

istioctl kube-inject \
    --filename common/alpine.yml \
    | kubectl apply --filename -

kubectl get pods

kubectl describe pod \
    --selector app=alpine

kubectl delete \
    --filename common/alpine.yml

###############################
# Automatic Sidecar Injection #
###############################

kubectl apply \
    --filename common/alpine.yml

kubectl get pods

kubectl label namespace default \
    istio-injection=enabled

kubectl describe namespace default

kubectl rollout restart \
    deployment alpine

kubectl get pods

kubectl describe pod \
    --selector app=alpine

kubectl label namespace default \
    istio-injection-

kubectl rollout restart \
    deployment alpine

kubectl get pods

kubectl delete \
    --filename common/alpine.yml

###############
# Cleaning Up #
###############

istioctl manifest generate \
    --set profile=demo \
    | kubectl delete -f -

cd ..

# Destroy the cluster (optional)
eksctl delete cluster kombs-eks
