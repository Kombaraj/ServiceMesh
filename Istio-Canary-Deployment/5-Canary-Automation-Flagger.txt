######################
# Installing Flagger #
# Flagger is a progressive delivery tool that automates the release process for applications running on Kubernetes. 
# It reduces the risk of introducing a new software version in production by gradually shifting traffic to the new version 
# while measuring metrics and running conformance tests.
######################

kubectl version --output yaml

# Must have `kubectl` v1.14+

kubectl apply \
    --kustomize github.com/weaveworks/flagger/kustomize/istio

kubectl get crd | grep flagger

#####################
# Deploying The App #
#####################

cd code

kubectl create namespace go-app

kubectl label namespace go-app \
    istio-injection=enabled


kubectl --namespace go-app apply \
    --filename istio/hpa \
    --recursive

kubectl --namespace go-app \
    rollout status \
    deployment go-app


kubectl --namespace go-app \
    get deploy,pods,services,gateway,virtualservices,hpa

# If EKS
export INGRESS_HOST=$(kubectl \
    --namespace istio-system \
    get service istio-ingressgateway \
    --output jsonpath="{.status.loadBalancer.ingress[0].hostname}")

echo $INGRESS_HOST

curl -H "Host: go-app.kombaraj.com" \
        "http://$INGRESS_HOST/api/hello"


##############################
# Deploying Flagger Resource #
##############################

cat istio/flagger/exercise/flagger.yaml

kubectl --namespace go-app apply \
    --filename istio/flagger/exercise/flagger.yaml

kubectl --namespace go-app \
    get deploy,pods,services,hpa,gateway,virtualservices,destinationrules,canaries


# Might be `<unknown>` if metrics server is not installed

for i in {1..10}; do 
    curl -H "Host: go-app.kombaraj.com" \
        "http://$INGRESS_HOST/api/hello"
    sleep 0.25
done

kubectl --namespace go-app \
    describe canary go-app

###########################
# Deploying A New Release v2.0.0 #
###########################

cat istio/flagger/exercise/go-app-2-0-0.yaml

# Open a second terminal session

# If EKS
export INGRESS_HOST=$(kubectl \
    --namespace istio-system \
    get service istio-ingressgateway \
    --output jsonpath="{.status.loadBalancer.ingress[0].hostname}")

echo $INGRESS_HOST

while true; do 
    curl -H "Host: go-app.kombaraj.com" \
        "http://$INGRESS_HOST/api/hello"
    sleep 1
done

# Go back to the first terminal session

kubectl --namespace go-app apply \
    --filename istio/flagger/exercise/go-app-2-0-0.yaml

kubectl --namespace go-app \
    get deploy,pods,services,hpa,gateway,virtualservices,destinationrules,canaries

kubectl --namespace go-app \
    describe virtualservice go-app

kubectl --namespace go-app \
    get canaries

kubectl --namespace go-app \
    describe canary go-app

kubectl --namespace go-app \
    describe virtualservice go-app

kubectl --namespace go-app \
    get canaries --watch
