#################
# Install Istio #
#################

istioctl install --set profile=demo -y

#################
# Using Gateway #
#################

cd code

ls -1 istio/gateway/

cat istio/gateway/app/istio.yaml

kubectl create namespace go-app

kubectl label namespace go-app \
    istio-injection=enabled

kubectl --namespace go-app apply \
    --filename istio/gateway \
    --recursive

kubectl --namespace go-app \
    rollout status \
    deployment go-app-primary

kubectl --namespace go-app \
    get deploy,pods,service,virtualservices,gateway

# Consume service from another pod
kubectl run curl \
    --image alpine \
    -it --rm \
    -- sh -c "apk add -U curl && curl go-app.go-app/api/hello"

kubectl --namespace go-app \
    get ingress

kubectl --namespace go-app \
    get gateways

kubectl --namespace go-app\
    describe gateway go-app

# If EKS
export INGRESS_HOST=$(kubectl \
    --namespace istio-system \
    get service istio-ingressgateway \
    --output jsonpath="{.status.loadBalancer.ingress[0].hostname}")

echo $INGRESS_HOST

# Consume the service from outside
curl -v -H "Host: go-app.kombaraj.com" \
    "http://$INGRESS_HOST/api/hello"

curl -v -H "Host: something-else.dummy.com" \
    "http://$INGRESS_HOST/api/hello"


kubectl --namespace go-app delete \
    --filename istio/gateway \
    --recursive

###############
# Cleaning Up #
###############

cd ..

kubectl delete namespace go-app

# Uninstall Istio
istioctl manifest generate \
    --set profile=demo \
    | kubectl delete -f -
    
# Destroy the cluster (optional)
eksctl delete cluster eksdemo1
