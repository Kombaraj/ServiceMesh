
####################
# Creating Cluster #
####################

######################
# Install Flagger #
######################

kubectl apply \
    --kustomize github.com/weaveworks/flagger/kustomize/istio

kubectl get crd | grep flagger

# Install Prometheus
kubectl apply -f https://raw.githubusercontent.com/istio/istio/release-1.11/samples/addons/prometheus.yaml

#####################
# Deploying The App #
#####################

cd code

kubectl create namespace go-app

kubectl label namespace go-app \
    istio-injection=enabled

kubectl --namespace go-app apply \
    --filename istio/flagger/app-db/db \
    --recursive

kubectl --namespace go-app apply \
    --filename istio/flagger/app-db/app \
    --recursive

kubectl --namespace go-app \
    rollout status \
    deployment go-app

kubectl --namespace go-app \
    rollout status \
    deployment go-app-primary

kubectl --namespace go-app \
    get deploy,pods,services,hpa,gateway,virtualservices,destinationrules,canaries

# If EKS
export INGRESS_HOST=$(kubectl \
    --namespace istio-system \
    get service istio-ingressgateway \
    --output jsonpath="{.status.loadBalancer.ingress[0].hostname}")

echo $INGRESS_HOST

curl -H "Host: go-app.kombaraj.com" \
    "http://$INGRESS_HOST/version"

##########################
# Rolling Back On Errors #
##########################

cat istio/flagger/app-db/flagger-error.yaml

kubectl --namespace go-app apply \
    --filename istio/flagger/app-db/flagger-error.yaml

# Open a second terminal session

export INGRESS_HOST=$(kubectl \
    --namespace istio-system \
    get service istio-ingressgateway \
    --output jsonpath="{.status.loadBalancer.ingress[0].hostname}")

echo $INGRESS_HOST

while true; do 
    curl -H "Host: go-app.kombaraj.com" \
        "http://$INGRESS_HOST/demo/random-error"
    sleep 0.5
done

# Go back to the first terminal session

kubectl --namespace go-app apply \
    --filename istio/flagger/app-db/deployment-0-0-2.yaml

kubectl --namespace go-app \
    describe canary go-app

# Go to the second terminal session and stop the loop with *ctrl+c*

########################################
# Rolling Back On Max Request Duration #
########################################

# Go to the first terminal session

cat istio/flagger/app-db/flagger-max-req-duration.yaml

kubectl --namespace go-app apply \
    --filename istio/flagger/app-db/flagger-max-req-duration.yaml

# Go to second terminal session

while true; do 
    DELAY=$[ $RANDOM % 3000 ]
    curl -H "Host: go-app.kombaraj.com" \
        "http://$INGRESS_HOST/demo/hello?delay=$DELAY"
done

# Go back to the first terminal session

kubectl --namespace go-app apply \
    --filename istio/flagger/app-db/deployment-0-0-3.yaml

kubectl --namespace go-app \
    describe canary go-app

# Go to the second terminal and stop the loop with *ctrl+c*

###################
# Rolling Forward #
###################

# Go to second terminal session

while true; do 
    curl -H "Host: go-app.kombaraj.com" \
        "http://$INGRESS_HOST/demo/hello"
    sleep 0.5
done

# Go back to the first terminal session

kubectl --namespace go-app apply \
    --filename istio/flagger/app-db/deployment-0-0-4.yaml

kubectl --namespace go-app \
    describe canary go-app
    
kubectl --namespace go-app \
    get deploy,pods,services,hpa,gateway,virtualservices,destinationrules,canaries


# Go to the second terminal and stop the loop with *ctrl+c*

cd ..

kubectl delete namespace go-app

# Uninstall Istio
istioctl manifest generate \
    --set profile=demo \
    | kubectl delete -f -

# Destroy the cluster (optional)
eksctl delete cluster eksdemo1
